"""PathLike shim for the source-inventory-pinned continuation stages only."""
from pathlib import Path
import argparse, json, runpy, sys

def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--repo',required=True,type=Path)
    parser.add_argument('--allowlist-json',required=True,type=Path)
    parser.add_argument('--script',required=True,type=Path)
    parser.add_argument('arguments',nargs=argparse.REMAINDER)
    args=parser.parse_args()
    repo=args.repo.resolve(); script=args.script.resolve()
    plan=json.loads(args.allowlist_json.read_text(encoding='utf-8'))
    pinned_hashes=set(plan['source_inventory'].values())
    import hashlib
    digest=hashlib.sha256(script.read_bytes()).hexdigest()
    if digest not in pinned_hashes:
        raise PermissionError('Only bytes represented by the source-inventory pins may execute')
    sys.path[:0]=[str(repo/'src'),str(repo),str(repo/'scripts')]
    import paired_extended_v1 as extended
    from yuntapr.contracts.loader import sha256 as original_sha256
    extended.sha256=lambda value: original_sha256(Path(value))
    remainder=args.arguments[1:] if args.arguments[:1]==['--'] else args.arguments
    sys.argv=[str(script),*remainder]
    runpy.run_path(str(script),run_name='__main__')

if __name__=='__main__': main()
