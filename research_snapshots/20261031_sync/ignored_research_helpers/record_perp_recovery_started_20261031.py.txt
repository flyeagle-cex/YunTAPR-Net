def read(p):
    return json.loads(Path(p).read_bytes())

def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def query(p, sql):
    c = sqlite3.connect(Path(p).resolve().as_uri() + '?mode=ro', uri=True)
    try:
        return c.execute(sql).fetchall()
    finally:
        c.close()
