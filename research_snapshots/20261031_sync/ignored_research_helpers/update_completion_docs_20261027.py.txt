def load(p):
    return p.read_bytes().decode('utf-8').replace('\r\n', '\n')

def save(p, s):
    previous = p.read_bytes()
    nl = '\r\n' if b'\r\n' in previous else '\n'
    p.write_bytes(s.replace('\r\n', '\n').replace('\n', nl).encode('utf-8'))

def replace(p, a, b):
