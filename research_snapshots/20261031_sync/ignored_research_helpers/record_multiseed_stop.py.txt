# Archival copy: formatting normalized; private locators redacted.

import json, sqlite3, hashlib, csv
from pathlib import Path
from datetime import datetime, timezone, timedelta
root = Path(__file__).resolve().parents[3]
local = root / 'docs/research_readiness/.local'
public = root / 'docs/research_readiness/20261025'

def read(p):
    return json.loads(p.read_text(encoding='utf-8'))

def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def digest(v):
    return hashlib.sha256(json.dumps(v, sort_keys=True, separators=(',', ':'), allow_nan=False).encode()).hexdigest()
zone = timezone(timedelta(hours=8))
now = datetime.now(zone).isoformat()
v = read(local / 'b3_b4_multiseed_preparation_20261026.json')
p = Path(v['output_root'])
proposal = read(public / 'evidence/b3_b4_multiseed_proposal.json')
stop = read(p / 'failure.json')
report = dict(status='STOP_NO_AUTOMATIC_RETRY', verified_at=now, training_commit='52a8e90dcea156382d7f2c510c5891c78e610329', stopped_at=datetime.fromtimestamp((p / 'failure.json').stat().st_mtime, zone).isoformat(), cause=dict(observed_log='forrtl: error (200): program aborting due to window-CLOSE event', worker_exit=2, actor='NOT_DETERMINABLE_FROM_AVAILABLE_LOG', numerical_or_OOM_error='NOT_OBSERVED_IN_STOP_LOG', launcher_defect='Worker Popen omitted CREATE_NO_WINDOW although controller was detached', causal_limit='Window-close abort is observed; omission is a console-exposure defect, not proof of who or what closed the window'), package_failure=stop, tasks=[], data_version=proposal['data'], budget=proposal['budget'], sources={'package/failure.json': sha(p / 'failure.json')}, read_only_diagnosis=dict(formal_optimizer_updates=0, raw_weather_payload_reads=0, model_inference=0, checkpoint_deserializations=0, checkpoint_file_byte_hashes=['B3_s2030_E9_LAST', 'B4_s2030_E1_LAST']), fix=dict(status='IMPLEMENTED_NOT_USED_FOR_RESTART', change='Set CREATE_NO_WINDOW on each Windows training worker', training_math_changed=False, checks=dict(passed=3, seconds=0.14, actual_Windows_child_GetConsoleWindow=None, serial_order='PASS_SYNTHETIC', failure_stops_dependents='PASS_SYNTHETIC'), monitor='Package native abort now marks the interrupted task FAILED even without task failure.json; only owned monitor restarted', training_restart=False, original_execution_copy_modified=False), frozen_2025=True)
total_updates = total_files = total_bytes = total_reserved = 0
for kind, seed in [('B3', 2030), ('B4', 2030), ('B3', 2031), ('B4', 2031)]:
    d = p / f'{kind}_s{seed}'
    if not d.exists():
        report['tasks'].append(dict(model=kind, seed=seed, status='NOT_STARTED', updates=0, epochs_complete=0))
        continue
    q = d / 'updates.sqlite'
    db = sqlite3.connect(q.as_uri() + '?mode=ro', uri=True)
    rows = db.execute('SELECT model,state,COUNT(*),MIN(u),MAX(u) FROM updates GROUP BY model,state').fetchall()
    db.close()
    n = sum((r[2] for r in rows if r[1] == 'RETURNED'))
    assert rows == [(kind, 'RETURNED', n, 1, n)]
    sealed = []
    metrics = []
    passes = []
    files = delivered = reserved = 0
    states = {}
    for epoch in range(1, 10):
        rec = d / f'{kind}_LAST_epoch_{epoch:02d}.receipt.json'
        if rec.exists():
            train = read(d / f'{kind}_train_{epoch:02d}.json')
            dev = read(d / f'{kind}_development_{epoch:02d}.json')
            r = read(rec)
            assert r['updates'] == train['updates'] == 480 * epoch and r['code_commit'] == report['training_commit'] and (r['validation_receipt_sha256'] == digest(dev))
            assert (d / f'{kind}_LAST_epoch_{epoch:02d}.pt').is_file()
            sealed.append(r)
            s = dev['groups']['all']['summary']
            metrics.append(dict(epoch=epoch, **{k: s[k] for k in ('brier', 'auroc', 'average_precision', 'conditional_pinball', 'n_valid', 'n_rain')}, q32_coverage=s['coverage'][-1]))
            for f in (rec, d / f'{kind}_train_{epoch:02d}.json', d / f'{kind}_development_{epoch:02d}.json'):
                report['sources'][f'{kind}_s{seed}/{f.name}'] = sha(f)
        for role in ('train', 'dev'):
            folder = d / f'{kind}_{role}_reads_{epoch:02d}'
            q = folder / 'reads.sqlite'
            if not q.exists():
                continue
            db = sqlite3.connect(q.as_uri() + '?mode=ro', uri=True)
            counts = db.execute('SELECT COUNT(*),COALESCE(SUM(actual_bytes),0),COALESCE(SUM(expected_bytes),0) FROM file_io').fetchone()
            group = dict(db.execute('SELECT state,COUNT(*) FROM file_io GROUP BY state').fetchall())
            db.close()
            for k, value in group.items():
                states[k] = states.get(k, 0) + value
            files += counts[0]
            delivered += counts[1]
            reserved += counts[2]
            seal = folder / 'pass_receipt.json'
            row = dict(epoch=epoch, role=role, files=counts[0], delivered_bytes=counts[1], reserved_bytes=counts[2], states=group, sealed=seal.exists())
            if seal.exists():
                sr = read(seal)
                scope = proposal['scopes']['Train' if role == 'train' else 'Development']
                assert not sr['during_failure'] and counts[0] == sr['expected_files'] == scope['file_count']
                assert counts[1] == sr['expected_bytes'] == scope['bytes_per_pass'] and sr['source_manifest_sha256'] == scope['source_manifest_sha256']
                assert group == {'READ_COMPLETE': counts[0]}
                report['sources'][f'{kind}_s{seed}/{folder.name}/pass_receipt.json'] = sha(seal)
            passes.append(row)
    latest = sealed[-1]
    weight = d / f"{kind}_LAST_epoch_{latest['epoch']:02d}.pt"
    assert sha(weight) == latest['checkpoint_sha256']
    complete = (d / 'completion.json').exists()
    if complete:
        assert kind == 'B3' and n == 4320 and (len(sealed) == 9) and (len(passes) == 18) and (delivered == proposal['budget']['source_bytes_per_model'])
        report['sources'][f'{kind}_s{seed}/completion.json'] = sha(d / 'completion.json')
    report['sources'][f'{kind}_s{seed}/updates.sqlite'] = sha(d / 'updates.sqlite')
    task = dict(model=kind, seed=seed, status='COMPLETE_TRAIN_AND_DEV_EVALUATION' if complete else 'STOPPED_INCOMPLETE', updates=n, ledger_rows=rows, epochs_complete=len(sealed), checkpoint_receipts=sealed, latest_checkpoint_file_sha256_verified=latest['checkpoint_sha256'], latest_checkpoint_updates=latest['updates'], uncheckpointed_returned_updates=n - latest['updates'], epochs=metrics, io=dict(files=files, delivered_bytes=delivered, reserved_bytes=reserved, states=states, passes=passes))
    report['tasks'].append(task)
    total_updates += n
    total_files += files
    total_bytes += delivered
    total_reserved += reserved
report['actual_totals'] = dict(returned_updates=total_updates, optimizer_INTENT=0, source_files=total_files, delivered_source_bytes=total_bytes, reserved_source_bytes=total_reserved, uncertain_source_entries=1, note='One B4 E2 source INTENT remains; do not relabel its source read successful or reread it under consumed approval')
report['recovery_boundary'] = dict(status='NOT_AUTHORIZED_NOT_EXECUTED', latest_B4_checkpoint_updates=480, uncheckpointed_B4_updates=390, original_replay_allowance=0, remaining_original_update_cap=17280 - total_updates, updates_needed_from_E1_for_remaining_three_endpoints=3840 + 8640, hypothetical_replay_over_original_update_cap=390, note='No restart or automatic replay; E1 restart would repeat390 returned updates and require a new explicit recovery decision')
out = public / 'evidence/b3_b4_multiseed_stop.json'
assert not out.exists()
out.write_text(json.dumps(report, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
matrix = public / 'B0_B4_EXPERIMENT_MATRIX.csv'
old = matrix.read_bytes()
lines = old.splitlines(keepends=True)
with matrix.open(encoding='utf-8-sig', newline='') as f:
    reader = csv.DictReader(f)
    fields = reader.fieldnames
    rows = list(reader)
import io
for i, row in enumerate(rows):
    if row['seed'] not in ('2030', '2031'):
        continue
    task = next((t for t in report['tasks'] if t['model'] == row['model_id'] and str(t['seed']) == row['seed']))
    row.update(status=task['status'], epochs=str(task['epochs_complete']), global_steps=str(task['updates']), evidence_path='evidence/b3_b4_multiseed_stop.json', verified_at=now, authorization_status='PACKAGE_STOPPED_NO_RESTART_AUTHORIZED', notes='Native window-CLOSE abort stopped B4; B3 complete; no automatic replay. Snapshot counters are not live.')
    if task['updates']:
        row['executed_commit'] = report['training_commit']
        row['checkpoint_reference'] = task['latest_checkpoint_file_sha256_verified']
        if row['model_id'] == 'B3':
            row['metrics_path'] = 'evidence/b3_b4_multiseed_stop.json#tasks/0/epochs/8'
            row['authorization_status'] = 'APPROVED_RUN_COMPLETE'
        else:
            row['notes'] = 'E1 checkpoint at480;870 returned;390 uncheckpointed. E9 endpoint NOT_AVAILABLE; one source INTENT preserved.'
    stream = io.StringIO(newline='')
    writer = csv.DictWriter(stream, fields)
    writer.writerow(row)
    lines[i + 1] = stream.getvalue().encode('utf-8')
matrix.write_bytes(b''.join(lines))
print(json.dumps(dict(status='STOP_RECORDED_NO_RESTART', updates=total_updates, bytes=total_bytes, source_files=total_files, receipt_sources=len(report['sources']))))
