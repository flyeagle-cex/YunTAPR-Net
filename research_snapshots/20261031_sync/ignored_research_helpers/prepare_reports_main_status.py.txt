def replace(path, old, new):
    b = path.read_bytes()
    assert old.encode() in b, (str(path), old[:40])
    path.write_bytes(b.replace(old.encode(), new.encode(), 1))

def insert(path, s):
    b = path.read_bytes()

    if i == 1:

    path.write_bytes(b[:i] + s.encode() + b[i:])
