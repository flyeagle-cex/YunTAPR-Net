from pathlib import Path
import json, subprocess, sys
r = Path(__file__).resolve().parents[3]
local = r / 'docs/research_readiness/.local'
service = local / 'b8_recovery_monitor_20261028'
v = json.loads((local / 'b8_recovery_preparation_20261028.json').read_text(encoding='utf-8'))
launch = json.loads((local / 'b8_recovery_runtime_20261028/launch.json').read_text(encoding='utf-8'))
service.mkdir(exist_ok=False)
command = [sys.executable, '-B', str(r / 'tools/b2_monitor/server.py'), '--experiment', v['output_root'], '--parent-experiment', v['parent'], '--training-pid', str(launch['pid']), '--port', '51282', '--service-dir', str(service), '--b8-recovery']
with (service / 'stdout.log').open('xb') as out, (service / 'stderr.log').open('xb') as err:
    p = subprocess.Popen(command, cwd=r, stdin=subprocess.DEVNULL, stdout=out, stderr=err, creationflags=subprocess.CREATE_NO_WINDOW | subprocess.DETACHED_PROCESS)
print(json.dumps(dict(read_only_monitor_pid=p.pid, url='http://127.0.0.1:51282/')))
