def show_scope():
    with open(PREP, 'r', encoding='utf-8') as f:

    p = w['prepared']
    print('== top-level keys ==', sorted(w.keys()))
    print('== prepared keys ==', sorted(p.keys()))

    print('private_inputs:', json.dumps(p['private_inputs'], indent=2))
    print('preparation_sha256:', p.get('preparation_sha256'))
    for role in ('Train', 'Development'):
        s = p['roles'][role]
        print(f'== {role} ==')
        print(' ids:', len(s['ids']), 'first:', s['ids'][:4], 'last:', s['ids'][-2:])
        print(' file_count:', s['file_count'], 'bytes_per_pass:', s['bytes_per_pass'])
        print(' by_kind:', json.dumps(s['by_kind'], indent=2))
        print(' manifest_sha256:', s['source_manifest_sha256'])
        print(' unique files:', len(s['files']))
    total = sum((s['bytes_per_pass'] for s in p['roles'].values()))
    print('sum(bytes_per_pass) =', total, 'x9 =', 9 * total)
    print('sum(file_count) =', sum((s['file_count'] for s in p['roles'].values())), 'x9 =', 9 * sum((s['file_count'] for s in p['roles'].values())))
    dev = p['roles']['Development']
    kinds = {}
    for name, item in dev['files'].items():
        kinds.setdefault(item['kind'], [0, 0])
        kinds[item['kind']][0] += 1
        kinds[item['kind']][1] += item['bytes']
    print('Development kind mix:', json.dumps(kinds, indent=2))
    train = p['roles']['Train']
    kinds = {}
    for name, item in train['files'].items():
        kinds.setdefault(item['kind'], [0, 0])
        kinds[item['kind']][0] += 1
        kinds[item['kind']][1] += item['bytes']
    print('Train kind mix:', json.dumps(kinds, indent=2))
    print('mask sources:', [n for n in dev['files'] if 'mask' in n.lower()][:3])

def check_runs():
    print('== 19 main-table dev/checkpoint existence ==')
    bad = 0
    for model, seed, dev, ckpt in RUNS:
        d = os.path.exists(dev)
        c = os.path.exists(ckpt)
        bad += (not d) + (not c)
        print(f"{model}/s{seed}: dev={d} ckpt={c} dev_bytes={(os.path.getsize(dev) if d else '-')} ckpt_bytes={(os.path.getsize(ckpt) if c else '-')}")
    print('missing:', bad)
    with open(os.path.join(S2027, 'B0_MATCHED_V2__E0__s2027.json'), 'r', encoding='utf-8') as f:
        j = json.load(f)
    print('s2027 keys:', sorted(j.keys()))
    print('epochs:', len(j.get('epochs', [])))
    last = j['epochs'][-1]
    print('epoch keys:', sorted(last.keys()))
    print('development keys:', sorted(last['development'].keys()))
    print('summary keys:', sorted(last['development']['summary'].keys())[:12])
    for k in ('brier', 'auroc', 'average_precision', 'conditional_pinball', 'n_valid', 'n_rain'):
        print(' ', k, '=', last['development']['summary'][k])
