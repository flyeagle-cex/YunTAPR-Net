def git(args, cwd=REPO):
    r = subprocess.run(['git', '-C', cwd] + args, capture_output=True, text=True, check=False)
    return r.stdout.strip() if r.returncode == 0 else None

def hash_object(path):
    r = subprocess.run(['git', '-C', REPO, 'hash-object', path], capture_output=True, text=True, check=False)
    return r.stdout.strip() if r.returncode == 0 else None

def compare_commit_to_worktree(commit, worktree, files):
    out = {'commit': commit, 'worktree': worktree, 'n': len(files), 'mismatch': [], 'missing': []}
    for rel in files:
        blob = git(['rev-parse', f'{commit}:{rel}'])
        local = os.path.join(worktree, rel.replace('/', os.sep))
        if not os.path.exists(local):
            out['missing'].append(rel)
            continue
        if hash_object(local) != blob:
            out['mismatch'].append(rel)
    out['matched'] = out['n'] - len(out['mismatch']) - len(out['missing'])
    return out

def compare_pins_to_worktree(pins, worktree, label, only=None):
    selected = {rel: h for rel, h in pins.items() if only is None or rel in only}
    out = {'label': label, 'worktree': worktree, 'n': len(selected), 'n_total': len(pins), 'mismatch': [], 'missing': []}
    for rel, expected in selected.items():
        local = os.path.join(worktree, rel.replace('/', os.sep))
        if not os.path.exists(local):
            out['missing'].append(rel)
            continue
        h = hashlib.sha256(open(local, 'rb').read()).hexdigest()
        if h != expected:
            out['mismatch'].append(rel)
    out['matched'] = out['n'] - len(out['mismatch']) - len(out['missing'])
    return out
