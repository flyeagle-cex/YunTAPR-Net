def sha256_file(path, limit=None):
    h = hashlib.sha256()
    total = 0
    with open(path, 'rb') as f:
        while True:
            chunk = f.read(8 * 1024 * 1024)
            if not chunk:
                break
            h.update(chunk)
            total += len(chunk)
            if limit and total >= limit:
                break
    return h.hexdigest()

def main():
    out = {'runs': []}
    for model, seed, arm, dev, ckpt in RUNS:
        row = dict(model=model, seed=seed, arm=arm, dev_path=dev, checkpoint_path=ckpt)
        for tag, path in (('dev', dev), ('checkpoint', ckpt)):
            row[tag + '_exists'] = os.path.exists(path)
            if row[tag + '_exists']:
                row[tag + '_bytes'] = os.path.getsize(path)
                row[tag + '_sha256'] = sha256_file(path)
        out['runs'].append(row)
        print(json.dumps({k: row[k] for k in row if k.endswith(('_exists', '_bytes', '_sha256'))} | {'model': model, 'seed': seed}, ensure_ascii=False), flush=True)
    outdir = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), 'outputs')
    with open(os.path.join(outdir, 'probe_registry_sha.json'), 'w', encoding='utf-8') as f:
        json.dump(out, f, ensure_ascii=False, indent=2)
    print('WROTE', os.path.join(outdir, 'probe_registry_sha.json'))
