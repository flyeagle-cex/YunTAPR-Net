def load_stored_summary(loader, path):
    with open(path, 'r', encoding='utf-8') as f:
        j = json.load(f)
    if loader == 'dev_report':
        s = j['summary']
        identity = dict(source_sha=j.get('source_sha'), ids_sha=j.get('ids_sha'), scope=j.get('scope'))
    elif loader == 's2027':
        dev = j['epochs'][-1]['development']
        s = dev['summary']
        identity = dict(source_sha=dev.get('source_sha'), ids_sha=dev.get('ids_sha'), scope=dev.get('scope'))
    else:
        s = j['groups']['all']['summary']
        identity = dict(source_identity=j.get('source_identity'), scope=j.get('scope'), groups=list(j.get('groups', {}).keys()))
    return (s, identity)
