def git(args, cwd=REPO):
    return subprocess.run(['git', '-C', cwd] + args, capture_output=True, text=True, check=False)

def tree_of(ref):
    r = git(['rev-parse', ref])
    return r.stdout.strip() if r.returncode == 0 else None

def compare():
    result = {}
    worktree_trees = {}
    for name, path in WORKTREES.items():
        if not os.path.exists(path):
            worktree_trees[name] = None
            continue
        head = tree_of('HEAD') if path == REPO else git(['rev-parse', 'HEAD'], cwd=path).stdout.strip()
        src = git(['rev-parse', head + ':src/yuntapr']).stdout.strip()
        cfg = git(['rev-parse', head + ':config']).stdout.strip()
        worktree_trees[name] = dict(head=head, src=src, config=cfg)
    result['worktree_trees'] = worktree_trees
    commit_trees = {}
    for label, commits in EXEC.items():
        rows = []
        for c in commits:
            cs = git(['rev-parse', c + ':src/yuntapr']).stdout.strip()
            cc = git(['rev-parse', c + ':config']).stdout.strip()
            match = [n for n, t in worktree_trees.items() if t and t['src'] == cs and (t['config'] == cc)]
            match_src = [n for n, t in worktree_trees.items() if t and t['src'] == cs]
            rows.append(dict(commit=c, src_tree=cs, config_tree=cc, exact_worktrees=match, src_only_worktrees=match_src))
        commit_trees[label] = rows
        print(label, json.dumps(rows)[:600], flush=True)
    result['commits'] = commit_trees
    return result
