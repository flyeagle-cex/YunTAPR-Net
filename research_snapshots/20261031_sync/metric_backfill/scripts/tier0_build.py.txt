def sha256_file(path: str) -> str:
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(8 * 1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()

def load_summary(loader: str, path: str):
    with open(path, 'r', encoding='utf-8') as f:
        j = json.load(f)
    if loader == 'dev_report':
        return (j['summary'], {})
    if loader == 's2027':
        dev = j['epochs'][-1]['development']
        return (dev['summary'], {})
    if loader == 'stage4':
        return (j['groups']['all']['summary'], j['groups'])
    raise ValueError(loader)

def occurrence_from_bins(reliability):
    """Reliability bins: bin b = [b/10,(b+1)/10) for b<9, bin9=[0.9,1]. P>=0.5 -> bins 5..9."""
    if len(reliability) != 10:
        raise ValueError('expected 10 reliability bins, got %d' % len(reliability))
    n_total = sum((int(b['n']) for b in reliability))
    r_total = sum((int(b['rain']) for b in reliability))
    tp = sum((int(b['rain']) for b in reliability[5:]))
    fp = sum((int(b['n']) - int(b['rain']) for b in reliability[5:]))
    fn = r_total - tp
    tn = n_total - r_total - fp
    out = {'N_valid_bins': n_total, 'N_rain_bins': r_total, 'TP': tp, 'FP': fp, 'FN': fn, 'TN': tn, 'POD': tp / r_total if r_total else None, 'FAR': fp / (tp + fp) if tp + fp else None, 'CSI': tp / (tp + fp + fn) if tp + fp + fn else None}
    denom = (tp + fp) * (fp + tn) + (tp + fn) * (fn + tn)
    out['HSS'] = 2 * (tp * tn - fp * fn) / denom if denom else None
    return out

def strict_from_sqlite(path: str):
    """Contingency from the exact score histogram (block=-1).

    Returns strict P>0.5 counts, plus the counts sitting exactly on 0.5, so that
    P>=0.5 can be reconstructed as strict + boundary.
    """
    import sqlite3
    import struct
    con = sqlite3.connect('file:' + path + '?mode=ro&immutable=1', uri=True)
    try:
        raw = con.execute('SELECT score,pos,neg FROM scores WHERE block=-1').fetchall()
    finally:
        con.close()
    tp = fp = tp_eq = fp_eq = n_all = 0
    for s, p, n in raw:
        v = struct.unpack('>d', s)[0]
        n_all += p + n
        if v > 0.5:
            tp += p
            fp += n
        elif v == 0.5:
            tp_eq += p
            fp_eq += n
    return {'TP_strict': tp, 'FP_strict': fp, 'TP_at_0.5': tp_eq, 'FP_at_0.5': fp_eq, 'N_valid_sqlite': n_all}

def eval_dir_for(loader: str, path: str):

    base = os.path.dirname(path)
    name = os.path.basename(path)

    idx = name.split('_development_')[1].split('.')[0]
    d = os.path.join(base, '%s_evaluation_%s' % (model, idx), 'all.sqlite')
    return d if os.path.exists(d) else None

    def write_csv(path, data, fields=None):
