def read_csv(name):
    path = os.path.join(OUT, name)
    with open(path, 'r', encoding='utf-8-sig', newline='') as f:
        return list(csv.DictReader(f))

def num(v):
    if v is None or v == '':
        return None
    if isinstance(v, (int, float)):
        return v
    try:
        return float(v)
    except ValueError:
        return v

def write_table(ws, start_row, headers, rows, widths=None, number_formats=None, wrap_cols=()):
    for j, h in enumerate(headers, 1):
        c = ws.cell(row=start_row, column=j, value=h)
        c.font = HEAD_FONT
        c.fill = HEAD_FILL
        c.alignment = Alignment(horizontal='center', vertical='center', wrap_text=True)
        c.border = BORDER
    for i, row in enumerate(rows, 1):
        for j, h in enumerate(headers, 1):
            v = row.get(h)
            if isinstance(v, str):
                v = num(v)
            c = ws.cell(row=start_row + i, column=j, value=v)
            c.border = BORDER
            c.font = Font(size=9)
            c.alignment = Alignment(vertical='center', wrap_text=h in wrap_cols, horizontal='center' if not isinstance(v, str) else 'left')
            if number_formats and h in number_formats and isinstance(v, (int, float)):
                c.number_format = number_formats[h]
    if widths:
        for j, h in enumerate(headers, 1):
            ws.column_dimensions[get_column_letter(j)].width = widths.get(h, 14)
    ws.row_dimensions[start_row].height = 34
    return start_row + len(rows)

def note(ws, row, text, fill=NOTE_FILL, merge_to=16):
    c = ws.cell(row=row, column=1, value=text)
    c.font = SMALL
    c.alignment = Alignment(wrap_text=True, vertical='top')
    c.fill = fill
    ws.merge_cells(start_row=row, start_column=1, end_row=row, end_column=merge_to)
    ws.row_dimensions[row].height = 26
