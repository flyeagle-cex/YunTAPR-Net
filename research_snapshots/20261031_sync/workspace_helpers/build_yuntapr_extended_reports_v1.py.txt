def read(p):
    return json.loads(Path(p).read_text(encoding='utf8'))

def save(p, v):
    with Path(p).open('x', encoding='utf8', newline='\n') as f:
        json.dump(v, f, ensure_ascii=False, indent=2, allow_nan=False)
        f.write('\n')

def pin(p):
    p = Path(p)
    h = hashlib.sha256()
    with p.open('rb') as f:
        for b in iter(lambda: f.read(1048576), b''):
            h.update(b)
    return {'path': str(p), 'bytes': p.stat().st_size, 'sha256': h.hexdigest()}

def esc(v):
    return ''.join(({'\\': '\\textbackslash{}', '_': '\\_', '&': '\\&', '%': '\\%', '#': '\\#', '$': '\\$', '{': '\\{', '}': '\\}', '~': '\\textasciitilde{}', '^': '\\textasciicircum{}'}.get(c, c) for c in str(v)))

def value(v):
    if isinstance(v, (dict, list)):
        return json.dumps(v, ensure_ascii=False, allow_nan=False)
    if v is None:
        return '未定义 / not defined'
    return str(v)

def cell(v):
    s = value(v)
    if len(s) == 64 and all((c in '0123456789abcdef' for c in s)):
        return '\\texttt{' + s[:32] + '}\\newline\\texttt{' + s[32:] + '}'
    if ':\\' in s or ':/' in s:
        return '{\\footnotesize\\EvidenceFont\\raggedright ' + '\\allowbreak{}'.join((esc(c) for c in s.replace('\\', '/'))) + '\\par}'
    return esc(s)
PREAMBLE = '\\documentclass[UTF8,11pt,fontset=windows]{ctexart}\n\\usepackage[a4paper,margin=23mm]{geometry}\n\\usepackage{amsmath,booktabs,longtable,tabularx,array,hyperref}\n\\newfontfamily\\EvidenceFont{Microsoft YaHei}\n\\hypersetup{colorlinks=true,linkcolor=black,urlcolor=blue}\n\\setlength{\\parindent}{0pt}\\setlength{\\parskip}{5pt}\n\\renewcommand{\\arraystretch}{1.15}\n\\newcolumntype{Y}{>{\\raggedright\\arraybackslash}X}\n\\begin{document}\n'
COMMON = '2025 原始数据读取、像元读取和模型推理样本数均为 0；2025 Final Test 未执行且未授权。B2 正式训练未授权。统计术语 population 在中文中表述为“样本集合”。所有模型和 optimizer 二进制仅存 F 盘，Git 仅保存来源、代码、配置、身份与报告。'

def report(folder, name, title, sections, refs):
    folder = Path(folder)
    folder.mkdir(parents=True, exist_ok=True)
    md = '# ' + title + '\n\n' + COMMON + '\n\n'
    tex = PREAMBLE + '\\begin{center}{\\Large\\bfseries ' + esc(title) + '}\\end{center}' + '\n' + esc(COMMON) + '\n\n'
    for heading, paragraph, fields in sections:
        md += '## ' + heading + '\n\n' + paragraph + '\n\n'
        tex += '\\section*{' + esc(heading) + '}\n' + esc(paragraph) + '\n\n'
        if fields:
            md += '| 项目 | 实际记录 |\n|---|---|\n' + ''.join(('| ' + esc(k).replace('\\_', '_') + ' | ' + value(v).replace('|', ' / ') + ' |\n' for k, v in fields)) + '\n'
            tex += '\\begin{longtable}{@{}p{.30\\textwidth}p{.65\\textwidth}@{}}\\toprule 项目 & 实际记录\\\\\\midrule\\endhead' + '\n'
            tex += '\n'.join((esc(k).replace('\\_', '\\_\\allowbreak{}') + ' & ' + cell(v) + ' \\\\' for k, v in fields)) + '\n' + '\\bottomrule\\end{longtable}' + '\n'
    md += '## 可追溯证据\n\n'
    tex += '\\section*{可追溯证据}' + '\n'
    for p in refs:
        ref = pin(p)
        md += '- ' + ref['path'] + '\n  SHA256: ' + ref['sha256'] + '\n\n'
        tex += cell(ref['path']) + '\n\n' + cell(ref['sha256']) + '\n\n'
    tex += '\\end{document}' + '\n'
    if '人口' in md or '人口' in tex:
        raise ValueError('Statistical translation regression')
    for suffix, text in [('.md', md), ('.tex', tex)]:
        with (folder / (name + suffix)).open('x', encoding='utf8', newline='\n') as f:
            f.write(text)
    p = folder / (name + '.tex')
    xelatex = Path('<PRIVATE_LOCATOR_ac4183a30422>')
    for n in (1, 2):
        r = subprocess.run([str(xelatex), '--disable-installer', '-interaction=nonstopmode', '-halt-on-error', p.name], cwd=folder, capture_output=True, text=True, encoding='utf8', errors='replace')
        (folder / (name + f'.compile_pass_{n}.txt')).write_text(r.stdout + r.stderr, encoding='utf8')
        if r.returncode:
            raise RuntimeError('LaTeX compilation failed: ' + str(p))
    return {'markdown': pin(p.with_suffix('.md')), 'tex': pin(p), 'pdf': pin(p.with_suffix('.pdf')), 'visual_review_pending': True}
