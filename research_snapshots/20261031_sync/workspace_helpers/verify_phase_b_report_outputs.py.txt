    return hashlib.sha256(p.read_bytes()).hexdigest()

    p.write_text(json.dumps(obj, ensure_ascii=False, indent=2) + '\n', encoding='utf-8', newline='\n')

for token in ('128,964', '23,447', '11,724', '254', '242', status['FINAL_CHECKPOINT_SHA256'], 'c7042beac2412594ca9fc264d7df10981755b393158c539cb48c9b56cd314327'):
    assert token in extracted, token
assert 'Overfull' not in (qa / (name + '.log')).read_text(encoding='utf-8', errors='replace')
pages = sorted(qa.glob('page-*.png'))
assert len(pages) == 3

    ast.parse((repo / 'scripts' / script).read_text(encoding='utf-8'))
proof = {'status': 'PASS', 'page_count': 3, 'LaTeX_compile_pass': True, 'overfull_boxes': 0, 'required_completion_identity_text_PASS': True, 'all_rendered_pages_visually_inspected': True, 'visual_result': 'Readable tables and formulas; no clipping or overlap; long FINAL path breaks at directory boundary', 'PDF_sha256': sha(out / (name + '.pdf')), 'editable_LaTeX_sha256': sha(out / (name + '.tex')), 'QA_directory': str(qa), 'rendered_page_sha256': {p.name: sha(p) for p in pages}, 'packages_installed_or_upgraded': False, 'verified_utc': datetime.now(timezone.utc).isoformat()}
