# Archival copy: formatting normalized; private locators redacted.

"""Publishable metadata archive/index; no raw/cache/model binary is admitted."""
from pathlib import Path
import datetime, gzip, hashlib, json, subprocess
BASE = Path(__file__).parent
REPO = BASE / 'YunTAPR-Net-push-chunks'

def read(p):
    return json.loads(Path(p).read_text(encoding='utf8'))

def save(p, v):
    with Path(p).open('x', encoding='utf8', newline='\n') as f:
        json.dump(v, f, ensure_ascii=False, indent=2, allow_nan=False)
        f.write('\n')

def sha(p):
    h = hashlib.sha256()
    with Path(p).open('rb') as f:
        for b in iter(lambda: f.read(1048576), b''):
            h.update(b)
    return h.hexdigest()
active = read(BASE / 'YunTAPR-extended-pipeline-active-v1.json')
run = Path(active['run_directory'])
summary = read(run / 'extended_summary_status.json')
assert summary['status'] == 'AUTHORIZED_NUMERICAL_PIPELINE_COMPLETE_REPORT_QA_AND_PUBLICATION_PENDING'
qa = read(run / 'pdf_visual_qa.json')
assert qa['all_pages_visually_verified']
old = read(active['phase_a_active_registry'])
oldrun = Path(old['run_directory'])
for rel, digest in read(oldrun / 'historical_file_snapshot.json')['files'].items():
    assert sha(REPO / rel) == digest, rel
inventory = read(run / 'sequential_execution/execution_plan.json')['source_inventory']
resume = read(BASE / 'YunTAPR-paired-resume-active-v1.json')
resume_inventory = read(Path(resume['resume_directory']) / 'sequential_execution/execution_plan.json')['source_inventory']
v2_plan = Path(resume['resume_directory']) / 'sequential_execution_v2/execution_plan.json'
v2_inventory = read(v2_plan)['source_inventory'] if v2_plan.exists() else {}
current_plan = Path(resume['pipeline_journal']) / 'execution_plan.json'
current_inventory = read(current_plan)['source_inventory'] if current_plan.exists() else {}
inventory = {**inventory, **resume_inventory, **v2_inventory, **current_inventory}
sources = run / 'orchestration_sources'
sources.mkdir(exist_ok=False)
copies = {}
helpers = [BASE / 'build_yuntapr_extended_reports_v1.py', BASE / 'archive_yuntapr_extended_delivery_v1.py', BASE / 'publish_yuntapr_extended_delivery_v1.py', BASE / 'update_yuntapr_extended_state_v1.py', BASE / 'register_yuntapr_extended_pipeline_v1.py', BASE / 'test_yuntapr_extended_drafts_v1.py', BASE / 'render_yuntapr_pipeline_reports_v1.py', BASE / 'monitor_yuntapr_extended_pipeline_v1.py']
helpers.extend([BASE / 'prepare_yuntapr_paired_resume_v1.py', BASE / 'refine_yuntapr_resume_counters_v1.py', BASE / 'create_yuntapr_resume_closure_v1.py', BASE / 'record_yuntapr_resume_preflight_history_v1.py', BASE / 'verify_yuntapr_real_resume_replay_v1.py', BASE / 'verify_yuntapr_resume_io_partition_v1.py', BASE / 'create_yuntapr_pipeline_fixture_adapter_v1.py', BASE / 'verify_yuntapr_inherited_fixture_scope_v1.py', BASE / 'observe_yuntapr_resumed_pipeline_v1.py', BASE / 'register_yuntapr_pipeline_controller_v2.py'])
helpers.extend([BASE / 'prepare_yuntapr_closure_path_recovery_v1.py', BASE / 'prepare_yuntapr_phase_b_path_dispatch_v1.py', BASE / 'verify_yuntapr_closure_path_recovery_v1.py', BASE / 'render_yuntapr_phase_a_reports_v1.py', BASE / 'record_yuntapr_phase_a_pdf_qa_v1.py', BASE / 'render_yuntapr_extended_ready_reports_v1.py', BASE / 'record_yuntapr_extended_stage_pdf_qa_v1.py', BASE / 'repair_yuntapr_review_path_layout_v1.py', BASE / 'repair_yuntapr_review_path_layout_v2.py', BASE / 'record_yuntapr_all_report_pdf_qa_v1.py'])
for source in sorted({Path(p) for p in inventory if Path(p).parent == BASE} | set(helpers)):
    digest = sha(source)
    if str(source) in inventory:
        assert digest == inventory[str(source)], 'Execution-time helper changed: ' + str(source)
    dest = sources / source.name
    dest.write_bytes(source.read_bytes())
    assert sha(dest) == digest
    copies[source.name] = {'sha256': digest, 'original_execution_path': str(source), 'role': 'Exact orchestration execution source; local workspace path layout and registry paths are required'}
save(sources / 'source_manifest.json', copies)
monitor_root = BASE / 'YunTAPR-live-monitor'
if monitor_root.exists():
    monitor_dest = sources / 'live_monitor'
    monitor_dest.mkdir(exist_ok=False)
    monitor_manifest = {}
    for name in ('server.py', 'processes.ps1', 'index.html', 'style.css', 'app.js', 'verify_ui.cjs'):
        source = monitor_root / name
        dest = monitor_dest / name
        dest.write_bytes(source.read_bytes())
        assert sha(dest) == sha(source)
        monitor_manifest[name] = {'sha256': sha(dest), 'original_execution_path': str(source)}
    for name in ('verification.json', 'source_integrity_verification.json'):
        source = monitor_root / 'runtime' / name
        if source.exists():
            dest = monitor_dest / name
            dest.write_bytes(source.read_bytes())
            assert sha(dest) == sha(source)
            monitor_manifest[name] = {'sha256': sha(dest), 'role': 'Read-only live monitor verification evidence'}
    save(monitor_dest / 'source_manifest.json', monitor_manifest)
execution = read(run / 'phase_b_execution_registry.json')
roots = [run, *[Path(v['public_directory']) for v in execution['formal_runs'].values()]]
archives = []
for root in roots:
    for source in sorted(root.rglob('*.jsonl')):
        dest = source.with_suffix('.jsonl.gz')
        assert not dest.exists(), str(dest)
        with source.open('rb') as r, dest.open('xb') as w, gzip.GzipFile(filename='', mode='wb', fileobj=w, mtime=0) as packed:
            for b in iter(lambda: r.read(1048576), b''):
                packed.write(b)
        h = hashlib.sha256()
        with gzip.open(dest, 'rb') as f:
            for b in iter(lambda: f.read(1048576), b''):
                h.update(b)
        assert h.hexdigest() == sha(source)
        archives.append({'original_local_path': str(source), 'original_sha256': sha(source), 'original_bytes': source.stat().st_size, 'publication_path': dest.relative_to(REPO).as_posix(), 'gzip_sha256': sha(dest), 'gzip_bytes': dest.stat().st_size, 'decompression_byte_identical': True, 'contains_only_metadata_text': True})
save(run / 'published_log_archive_manifest.json', archives)
prefixes = [p.relative_to(REPO).as_posix() + '/' for p in roots] + ['tests/paired_extended_pipeline/', 'tests/paired_phase_a_resume/', 'tests/paired_pipeline_fixture_scope/']
installed = read(run / 'installed_source_manifest.json')['files']
exact = set(installed) | {'config/training/b0_matched_phase_b_finalfit_extended_authorization_v1.json', 'config/training/b1_phase_b_finalfit_extended_authorization_v1.json'}
exact.add('scripts/paired_phase_a_resume_governance_v1.py')
exact.add('scripts/paired_pipeline_fixture_scope_v1.py')
exact.add('scripts/paired_pipeline_historical_inventory_v1.py')
exact.add('scripts/paired_extended_path_adapter_v1.py')
exact.update((p.relative_to(REPO).as_posix() for p in (REPO / 'config/training').glob('b1_phase_a_resume_authorization_run_*.json')))
untracked = subprocess.check_output(['git', 'ls-files', '--others', '--exclude-standard', '-z'], cwd=REPO).decode('utf8').split('\x00')[:-1]
files = {}
omitted = []
for rel in untracked:
    p = REPO / rel
    if rel not in exact and (not any((rel.startswith(prefix) for prefix in prefixes))):
        continue
    if p.suffix in ('.pt', '.pth', '.ckpt', '.nc', '.npy', '.npz', '.bin', '.safetensors'):
        raise ValueError('Forbidden binary in additive task subtree: ' + rel)
    if p.name == 'pipeline_state.json' or p.suffix in ('.jsonl', '.aux', '.out', '.log', '.synctex', '.pyc') or '.compile_pass_' in p.name:
        omitted.append({'path': rel, 'reason': 'mutable observer/build scratch/local metadata log with published gzip counterpart'})
        continue
    extra_monitor_asset = rel.startswith((sources / 'live_monitor').relative_to(REPO).as_posix() + '/') and p.suffix in ('.html', '.css', '.js', '.ps1', '.cjs')
    assert p.suffix in ('.py', '.json', '.csv', '.md', '.tex', '.pdf', '.txt', '.yaml', '.gz') or extra_monitor_asset, rel
    assert p.stat().st_size < 90 * 1048576, rel
    files[rel] = {'bytes': p.stat().st_size, 'sha256': sha(p)}
parent = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=REPO, text=True).strip()
save(run / 'delivery_scope_check.json', {'only_additions': True, 'no_model_optimizer_raw_cache_upload': True, 'historical_baseline_2061_byte_identical': True, 'phase_a_completed_snapshot_unchanged': True, '2025_RAW_ACCESS': 0, 'omitted_local_artifacts': omitted, 'metadata_gzip_count': len(archives)})
p = run / 'delivery_scope_check.json'
files[p.relative_to(REPO).as_posix()] = {'bytes': p.stat().st_size, 'sha256': sha(p)}
save(run / 'delivery_file_index.json', {'parent_commit': parent, 'files': files, 'only_additions': True, 'self_hash_excluded': 'delivery_file_index.json', 'total_bytes_without_index': sum((v['bytes'] for v in files.values()))})
print(json.dumps({'eligible_files': len(files) + 1, 'metadata_only_gzip_logs': len(archives), 'bytes_without_index': sum((v['bytes'] for v in files.values())), 'parent': parent}))
