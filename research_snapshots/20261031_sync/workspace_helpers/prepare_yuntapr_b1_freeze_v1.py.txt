from pathlib import Path
from datetime import datetime, timezone
import hashlib, json, subprocess, sys
ROOT = Path(__file__).parent
REPO = ROOT / 'YunTAPR-Net-push-chunks'
BASE = '1fd70ba9a563aa673012fa1b1494b86d72511eb6'
assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=REPO, text=True).strip() == BASE
assert not subprocess.check_output(['git', 'status', '--porcelain'], cwd=REPO)
history = {}
for entry in subprocess.check_output(['git', 'ls-tree', '-rz', BASE], cwd=REPO).split(b'\x00'):
    if not entry:
        continue
    meta, name = entry.split(b'\t', 1)
    name = name.decode('utf-8')
    data = (REPO / name).read_bytes()
    blob = hashlib.sha1(b'blob ' + str(len(data)).encode() + b'\x00' + data).hexdigest()
    assert blob == meta.decode().split()[2], name
    history[name] = {'sha256': hashlib.sha256(data).hexdigest(), 'git_blob_sha1': blob, 'bytes': len(data)}
assert len(history) == 1941

    with p.open('x', encoding='utf-8', newline='\n') as f:
