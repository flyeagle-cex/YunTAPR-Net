def escape(text):
    return ''.join(({'\\': '\\textbackslash{}', '_': '\\_', '&': '\\&', '%': '\\%', '#': '\\#', '$': '\\$', '{': '\\{', '}': '\\}', '~': '\\textasciitilde{}', '^': '\\textasciicircum{}'}.get(c, c) for c in str(text)))

def number(value):
    return '--（未定义）' if value is None else repr(value)

def field_table(fields):
    rows = []
    for name, value in fields:
        s = str(value)
        if len(s) == 64 and all((c in '0123456789abcdef' for c in s)):
            value_tex = '\\texttt{' + s[:32] + '}\\newline\\texttt{' + s[32:] + '}'
        elif ':\\' in s or ':/' in s:
            value_tex = '{\\footnotesize\\path{' + s.replace('\\', '/') + '}}'
        else:
            value_tex = escape(value)
        rows.append(escape(name) + ' & ' + value_tex + ' \\\\')
    return '\\begin{tabularx}{\\textwidth}{@{}p{41mm}Y@{}}\\toprule' + '\n' + '\n'.join(rows) + '\n' + '\\bottomrule\\end{tabularx}'

def write(path, text):
    with path.open('x', encoding='utf8', newline='\n') as f:
        f.write(text)
