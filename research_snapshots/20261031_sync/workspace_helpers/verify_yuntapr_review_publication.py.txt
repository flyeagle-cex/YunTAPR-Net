def sha(path):
    with Path(path).open('rb') as handle:
        return hashlib.file_digest(handle, 'sha256').hexdigest()

def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8'))

def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)

        assert sha(REPO / relative) == expected, relative

    for identity in manifest['formal_identity'].values():

            path = REPO / path
        assert sha(path) == identity['sha256'], str(path)
    best = manifest['formal_best']
    assert Path(best['absolute_local_path']).stat().st_size == best['bytes']
    assert sha(best['absolute_local_path']) == best['sha256']
    assert best['epoch'] == 11 and best['global_update'] == 64460

def audit_changes(staged=False, commit=None):
    args = ['diff', '--name-only', '-z']
    args += [PARENT, commit] if commit else ['--cached']
    paths = [p.decode('utf-8') for p in git(*args).split(b'\x00') if p]
    assert paths, 'Nothing staged or committed'
    files = {}
    for relative in paths:
        assert relative in EDITABLE or relative.startswith(PREFIXES), relative
        assert not relative.startswith(OLD)
        path = REPO / relative
        assert path.is_file() and path.suffix.lower() in {'.py', '.md', '.json', '.csv', '.txt', '.log', '.png', '.nc'}, relative
        data = path.read_bytes()
        assert len(data) <= (10000000 if path.suffix == '.csv' else 2000000), relative
        if path.suffix in {'.py', '.md', '.json', '.csv', '.txt', '.log'}:
            assert b'\x00' not in data, relative
            data.decode('utf-8')
        target = commit if commit else ''
        assert git('show', target + ':' + relative) == data, relative
        files[relative] = {'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest(), 'git_blob_sha1': hashlib.sha1(b'blob ' + str(len(data)).encode() + b'\x00' + data).hexdigest()}
    return files

def verify_remote(commit, files):
    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}))

    def get(suffix):
        request = urllib.request.Request('https://api.github.com/repos/flyeagle-cex/YunTAPR-Net/' + suffix, headers={'User-Agent': 'YunTAPR-scientific-review-publication-audit', 'Accept': 'application/vnd.github+json'})
        with opener.open(request, timeout=45) as response:
            return json.load(response)
    remote = get('commits/main')
    assert remote['sha'] == commit
    assert remote['parents'][0]['sha'] == PARENT
    tree = get('git/trees/' + commit + '?recursive=1')
    assert not tree.get('truncated')
    entries = {v['path']: v for v in tree['tree']}
    for relative, local in files.items():
        assert entries[relative]['sha'] == local['git_blob_sha1'], relative
        assert entries[relative]['size'] == local['bytes'], relative
    return {'status': 'PASS', 'main_commit': commit, 'github_blob_identities_verified': len(files)}
if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--staged', action='store_true')
    parser.add_argument('--remote', action='store_true')
    args = parser.parse_args()
    result = {'run_id': RUN_ID, 'local': audit_local(), 'checked_utc': datetime.now(timezone.utc).isoformat()}
    if args.staged or args.remote:
        commit = git('rev-parse', 'HEAD').decode().strip() if args.remote else None
        files = audit_changes(args.staged, commit)
        result['files'] = files
        result['publication_bytes'] = sum((v['bytes'] for v in files.values()))
        if args.remote:
            result['remote'] = verify_remote(commit, files)
        receipt = ROOT / ('YunTAPR-scientific-review-publication-' + (commit[:12] if commit else 'staged') + '.json')
        receipt.write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
        result = {k: v for k, v in result.items() if k != 'files'}
        result['receipt'] = str(receipt)
    print(json.dumps(result, ensure_ascii=False))
