def read(path):
    return json.loads(path.read_text(encoding='utf-8-sig'))

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)

def blob(data):
    return hashlib.sha1(b'blob ' + str(len(data)).encode() + b'\x00' + data).hexdigest()

def finalize_artifacts():
    qa = ROOT / 'tmp/pdfs/b1_final_design_v1'
    log = (qa / (NAME + '.log')).read_text(encoding='utf-8')
    text = (qa / 'text.txt').read_text(encoding='utf-8')
    assert 'Overfull' not in log and 'Missing character' not in log and ('人口' not in text)
    images = sorted(qa.glob('page-*.png'))
    assert len(images) == 6
    status = read(RUN / 'final_status.json')
    for digest in (status['definition_sha256'], status['frame_identity_index_sha256']):
        assert digest in text
    assert '23,520' in text and '70,560' in text and ('141,120' in text)
    target = DOC / (NAME + '.pdf')
    assert not target.exists()
    shutil.copyfile(qa / (NAME + '.pdf'), target)
    assert sha(target) == sha(qa / (NAME + '.pdf'))
    assert target.read_bytes().startswith(b'%PDF-')
    gate.save(RUN / 'render_verification.json', {'status': 'PASS', 'pages': 6, 'all_rendered_pages_visually_inspected': True, 'no_visible_clipping_or_overlap': True, 'overfull_boxes': 0, 'missing_glyphs': 0, 'forbidden_population_mistranslation_present': False, 'native_editor_tools_available': False, 'tool': 'Existing XeLaTeX --disable-installer + Poppler', 'packages_installed_or_upgraded': False, 'source_sha256': sha(target.with_suffix('.tex')), 'pdf_sha256': sha(target), 'rendered_page_sha256': {p.name: sha(p) for p in images}, 'verified_utc': gate.now()})
    helpers = ('prepare_yuntapr_b1_design.py', 'inspect_yuntapr_b1_prior_evidence.py', 'count_yuntapr_b1_candidate_parameters.py', 'close_yuntapr_b1_temporal_audit_tests.py', 'build_yuntapr_b1_final_decision_packet.py')
    gate.save(RUN / 'orchestration_provenance.json', {'helper_sha256': {n: sha(ROOT / n) for n in helpers}, 'helpers_are_local_orchestration_only': True, 'published_entrypoints': ['scripts/audit_b1_temporal_v1.py', 'scripts/count_b1_frontend_candidate_v1.py'], 'raw_execution_code_sha256': audit_module.code_hashes(), 'created_utc': gate.now()})
    names = NEW | {p.relative_to(REPO).as_posix() for root in (RUN, DOC, SOURCE_RUN) for p in root.rglob('*') if p.is_file()}
    names = {n for n in names if '__pycache__' not in Path(n).parts}
    names.discard(INDEX.relative_to(REPO).as_posix())
    gate.save(INDEX, {n: {'bytes': (REPO / n).stat().st_size, 'sha256': sha(REPO / n)} for n in sorted(names)})
    return {'new_artifacts_indexed': len(names), 'index': str(INDEX)}

def changes(commit=None):
    args = ('diff', '--name-status', '-z', BASE, commit) if commit else ('diff', '--cached', '--name-status', '-z')
    parts = [p.decode('utf-8') for p in git(*args).split(b'\x00') if p]
    result = {}
    prefixes = ('docs/b1_scientific_design/', 'docs/b1_temporal_audit/runs/' + RUN.name + '/', 'docs/b1_temporal_audit/runs/' + SOURCE_RUN.name + '/')
    for i in range(0, len(parts), 2):
        action, name = parts[i:i + 2]
        assert action == 'A' and (name in NEW or name.startswith(prefixes)), (action, name)
        path = REPO / name
        data = path.read_bytes()
        assert path.suffix in ('.py', '.md', '.tex', '.pdf', '.json', '.txt', '.csv') and len(data) < 20000000, name
        if path.suffix == '.pdf':
            assert data.startswith(b'%PDF-')
        else:
            assert b'\x00' not in data
            data.decode('utf-8')
            assert not re.search(b'ghp_[A-Za-z0-9]{30,}|github_pat_[A-Za-z0-9_]{20,}|BEGIN (?:RSA |EC )?PRIVATE KEY', data), name
        assert git('show', (commit or '') + ':' + name) == data, name
        result[name] = {'bytes': len(data), 'sha256': sha(path), 'blob_sha1': blob(data)}
    assert set(result) == set(read(INDEX)) | {INDEX.relative_to(REPO).as_posix()}
    return result

def remote(commit, files):
    previous.BASE = BASE
    return previous.remote(commit, files)
if __name__ == '__main__':
    parser = argparse.ArgumentParser()

    parser.add_argument('--staged', action='store_true')
    parser.add_argument('--remote', action='store_true')
    args = parser.parse_args()
    if args.finalize:

    else:
        result = {'utc': datetime.now(timezone.utc).isoformat(), 'local': audit()}
        if args.staged or args.remote:
            commit = git('rev-parse', 'HEAD').decode().strip() if args.remote else None
            files = changes(commit)
            result['files'] = files
            result['publication_bytes'] = sum((r['bytes'] for r in files.values()))
            if args.remote:
                result['remote'] = remote(commit, files)
            receipt = ROOT / ('YunTAPR-b1-design-publication-' + (commit[:12] if commit else 'staged') + '.json')
            receipt.write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
            result = {k: v for k, v in result.items() if k != 'files'}
            result['receipt'] = str(receipt)
        print(json.dumps(result))
